# 🛰️ Telemetry Stream Cleaner & Anomaly Detector

### 🎯 Para que foi criado
Limpar, normalizar e detectar anomalias em fluxos contínuos de telemetria dos sensores IoT (USGS Sísmico e Open-Meteo Clima) sem sobrecarregar a memória do servidor nem consumir tokens repetidos.

### 💡 Por que foi criado
A sociedade de agentes identificou que a ingestão contínua de streams meteorológicos e sísmicos brutos continha ruídos de rádio e oscilações pontuais de temperatura que gastavam tokens se analisados manualmente. Este sistema auto-contido em JavaScript e Canvas executa no cliente, aplicando janelas móveis de Z-Score para destacar eventos anômalos em tempo real.

### 📋 Requisitos & Especificações
- **Processamento**: Média móvel exponencial (EMA, alfa = 0.25) com janela de 50 amostras.
- **Detecção**: Limiar de Z-Score móvel configurável (padrão > 2.5 desvios padrão).
- **Visualização**: Gráficos de linha a 60fps renderizados em HTML5 Canvas nativo.
- **Exportação**: Suporte a download de dados processados em CSV e JSON com 1 clique.
- **Compatibilidade**: Zero bibliotecas externas pesadas; roda 100% offline em qualquer navegador.

  <script>
    const canvas = document.getElementById("chartCanvas");
    const ctx = canvas.getContext("2d");
    let isStreaming = true;
    let threshold = 2.5;
    let rawBuffer = [];
    let cleanBuffer = [];
    let anomaliesCount = 0;
    let totalSamples = 0;
    const maxPoints = 50;

    function resize() {
      canvas.width = canvas.parentElement.clientWidth - 32;
      canvas.height = 260;
    }
    window.addEventListener("resize", resize);
    resize();

    let curVal = 20.0;
    let ema = 20.0;
    const alpha = 0.25;

    function updateSensitivity(val) {
      threshold = parseFloat(val);
      document.getElementById("zVal").textContent = val + " σ";
    }

    function injectSpike() {
      curVal += (Math.random() > 0.5 ? 18.0 : -18.0);
    }

    function toggleStream() {
      isStreaming = !isStreaming;
      document.getElementById("btnStream").textContent = isStreaming ? "⏸️ Pausar Fluxo" : "▶️ Retomar";
      document.getElementById("streamStatus").textContent = isStreaming ? "🟢 PROCESSANDO A 60 FPS" : "🟡 PAUSADO";
      document.getElementById("streamStatus").style.color = isStreaming ? "#10b981" : "#f59e0b";
    }

    function step() {
      if (isStreaming) {
        totalSamples++;
        const noise = (Math.random() - 0.5) * 3.5;
        curVal = curVal * 0.95 + 20.0 * 0.05 + noise;
        ema = alpha * curVal + (1 - alpha) * ema;

        rawBuffer.push(curVal);
        cleanBuffer.push(ema);
        if (rawBuffer.length > maxPoints) {
          rawBuffer.shift();
          cleanBuffer.shift();
        }

        // Z-Score
        let mean = rawBuffer.reduce((a,b)=>a+b,0) / rawBuffer.length;
        let variance = rawBuffer.reduce((a,b)=>a+Math.pow(b-mean,2),0) / rawBuffer.length;
        let std = Math.sqrt(variance) || 1;
        let z = Math.abs(curVal - mean) / std;

        let isAnomaly = (z > threshold);
        if (isAnomaly) anomaliesCount++;

        document.getElementById("cntSamples").textContent = totalSamples;
        document.getElementById("valMean").textContent = ema.toFixed(2);
        document.getElementById("valStd").textContent = std.toFixed(2);
        document.getElementById("cntAnomalies").textContent = anomaliesCount;
      }

      // Render chart
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      const w = canvas.width;
      const h = canvas.height;

      // Draw grid
      ctx.strokeStyle = "#1e293b";
      ctx.lineWidth = 1;
      for (let i = 0; i < 5; i++) {
        let y = (h / 4) * i;
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(w, y); ctx.stroke();
      }

      if (rawBuffer.length > 1) {
        let stepX = w / (maxPoints - 1);
        let minVal = 0, maxVal = 40;

        // Raw signal (grey)
        ctx.strokeStyle = "#64748b";
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        rawBuffer.forEach((val, i) => {
          let x = i * stepX;
          let y = h - ((val - minVal) / (maxVal - minVal)) * h;
          if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
        });
        ctx.stroke();

        // Clean signal (cyan)
        ctx.strokeStyle = "#38bdf8";
        ctx.lineWidth = 2.5;
        ctx.beginPath();
        cleanBuffer.forEach((val, i) => {
          let x = i * stepX;
          let y = h - ((val - minVal) / (maxVal - minVal)) * h;
          if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
        });
        ctx.stroke();
      }

      requestAnimationFrame(step);
    }
    requestAnimationFrame(step);

    function exportCSV() {
      let csv = "Index,RawVal,CleanEMA\n";
      rawBuffer.forEach((r, i) => {
        csv += `${i},${r.toFixed(3)},${cleanBuffer[i].toFixed(3)}\n`;
      });
      const blob = new Blob([csv], { type: "text/csv" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = "telemetry_clean_stream.csv";
      a.click();
    }
