const CoreProcessor = {
  version: '1.0.0',
  author: 'deepseek-r1-7b-reasoning',

  calculateEntropy: function(dataStr) {
    if (!dataStr || dataStr.length === 0) return 0;
    const freq = {};
    for (let i = 0; i < dataStr.length; i++) {
      const c = dataStr[i];
      freq[c] = (freq[c] || 0) + 1;
    }
    let entropy = 0;
    const len = dataStr.length;
    for (let c in freq) {
      const p = freq[c] / len;
      entropy -= p * Math.log2(p);
    }
    return parseFloat(entropy.toFixed(4));
  },

  normalizeStream: function(values) {
    if (!Array.isArray(values) || values.length === 0) return [];
    const min = Math.min(...values);
    const max = Math.max(...values);
    if (max === min) return values.map(() => 0.5);
    return values.map(v => parseFloat(((v - min) / (max - min)).toFixed(4)));
  },

  processDataset: function(rawInput) {
    const t0 = performance.now();
    const lines = String(rawInput || '').split('\n').map(l => l.trim()).filter(Boolean);
    const numbers = [];
    lines.forEach(l => {
      const parts = l.split(/[,;\t\s]+/);
      parts.forEach(p => {
        const n = parseFloat(p);
        if (!isNaN(n)) numbers.push(n);
      });
    });
    const entropy = this.calculateEntropy(String(rawInput));
    const normalized = this.normalizeStream(numbers);
    const mean = numbers.length > 0 ? (numbers.reduce((a,b)=>a+b,0) / numbers.length).toFixed(3) : 0;
    const t1 = performance.now();
    return {
      items_count: numbers.length,
      lines_count: lines.length,
      entropy_bits: entropy,
      mean_value: parseFloat(mean),
      normalized_sample: normalized.slice(0, 50),
      latency_ms: parseFloat((t1 - t0).toFixed(2)),
      processed_by: 'CoreProcessor v1.0 (@deepseek-r1-7b-reasoning)'
    };
  }
};

    function executeProcess() {
      const val = document.getElementById("inputData").value;
      if (typeof CoreProcessor === "undefined") {
        document.getElementById("output").textContent = "Erro: CoreProcessor não carregado.";
        return;
      }
      const res = CoreProcessor.processDataset(val);
      document.getElementById("output").textContent = JSON.stringify(res, null, 2);
      renderChart(res.normalized_sample || []);
    }

    function renderChart(data) {
      const cvs = document.getElementById("chartCanvas");
      if (!cvs) return;
      const ctx = cvs.getContext("2d");
      cvs.width = cvs.parentElement.clientWidth - 36;
      cvs.height = 140;
      ctx.clearRect(0, 0, cvs.width, cvs.height);
      if (data.length === 0) return;

      ctx.beginPath();
      ctx.strokeStyle = "#38bdf8";
      ctx.lineWidth = 2;
      const step = cvs.width / Math.max(1, data.length - 1);
      data.forEach((d, idx) => {
        const x = idx * step;
        const y = cvs.height - (d * (cvs.height - 20) + 10);
        if (idx === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      });
      ctx.stroke();

      ctx.fillStyle = "#10b981";
      data.forEach((d, idx) => {
        const x = idx * step;
        const y = cvs.height - (d * (cvs.height - 20) + 10);
        ctx.beginPath();
        ctx.arc(x, y, 3, 0, Math.PI * 2);
        ctx.fill();
      });
    }

    function loadSample(type) {
      if (type === 1) {
        document.getElementById("inputData").value = "12.5, 14.2, 13.8, 15.1, 14.9, 16.2, 15.8, 29.4, 15.0, 14.8";
      } else {
        document.getElementById("inputData").value = "98.2, 1.4, 88.5, 3.2, 75.1, 2.9, 94.3, 1.1, 80.2, 4.5";
      }
      executeProcess();
    }

    window.addEventListener("load", () => {
      executeProcess();
    });
